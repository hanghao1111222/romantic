class RoseQuantumEmitter {
  constructor() {
    this.particles = 35000;
    this.spiralRev = 20 * Math.PI;
    this.bloomCoeff = 3.4;
  }

  evalSurface(t, x) {
    const p = (PI / 2) * exp(-t / (8 * PI));
    const u = 1 - pow(1 - (3.4 * t % 2PI) / PI, 4) / 2;
    const y = 2 * pow(x*x - x, 2) * sin(p);
    return [ u * (x*sin(p) + y*cos(p)), u * (x*cos(p) - y*sin(p)) ];
  }

  heartbeatPulse(time) {
    return 1.0 + 0.05 * sin(time * 2.8);
  }
}

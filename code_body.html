65# ==================== 2. 纯正数学 3D 玫瑰曲面点阵建模 ====================
66def generate_perfect_rose(total_points=4600):
67 """ 采用世界经典 3D 玫瑰参数曲面方程 (The Authentic 3D Rose Surface Equation) 精确计算花瓣卷曲螺旋、边缘褶皱、花托萼片与花茎 """
68 
69 
70
71 pal = PALETTES[current_theme]
72 particles = []
73
74 # 1. 核心花瓣曲面采样 (约 72% 粒子)
75 petal_points = int(total_points * 0.72)
76 for _ in range(petal_points):
77 # t 控制螺旋圈数与角度：从内侧花蕾到外层舒展大花瓣
78 t_norm = random.uniform(0, 1) ** 0.85
79 t = t_norm * (20.0 * math.pi) + 3.8 * math.pi
80
81 # p 控制花瓣从直立到外翻的俯仰倾角
82 p = (math.pi / 2.0) * math.exp(-t / (8.0 * math.pi))
83 # 边缘微波褶皱
84 ripple = math.sin(15.0 * t) / 140.0
85
86 # 每圈约 3.4 瓣花瓣，模运算营造自然的花瓣交叠重合
87 mod_t = (3.4 * t) % (2.0 * math.pi)
88 u_factor = 1.0 - ((1.0 - mod_t / math.pi) ** 4) / 2.0 + ripple
89
90 # x 为花瓣径向位置：0 为花瓣根部，1 为花瓣边缘
91 # 增加边缘采样概率，让花瓣优美的边缘轮廓清晰可辨！
92 if random.random() <0.35:
93 x = random.uniform(0.85, 1.0) # 强化边缘线条！
94 is_edge = True
95 else:
96 x = random.uniform(0.08, 0.95)
97 is_edge = False
98
99 # y profile 形成花瓣杯状凹陷弧度
100 y_prof = 2.0 * (x**2 - x) ** 2 * math.sin(p)
101
102 r = u_factor * (x * math.sin(p) + y_prof * math.cos(p))
103 h = u_factor * (x * math.cos(p) - y_prof * math.sin(p))
104
105 scale = 138.0
106 # 3D 空间坐标转换 (Y 向上，X/Z 展开)
107 base_x = r * math.cos(t) * scale
108 base_z = r * math.sin(t) * scale
109 base_y = (h * scale) - 20.0
110
111 # 确定着色
112 dist_c = math.sqrt(base_x**2 + base_z**2)
113
114 if is_edge:
115 col = random.choice(pal["edge"])
116 size = 2 if random.random() <0.4 else 1
117 elif dist_c <24 and base_y >-10:
118 col = random.choice(pal["core"]) # 黄金花心
119 size = 2 if random.random() <0.3 else 1
120 elif dist_c <65:
121 col = random.choice(pal["petal_high"])
122 size = 1
123 elif x <0.35 or base_y <-15:
124 col = random.choice(pal["petal_dark"])
125 size = 1
126 else:
127 col = random.choice(pal["petal_mid"])
128 size = 1
129
130 particles.append({
131 "x": base_x, "y": base_y, "z": base_z,
132 "col": col, "size": size, "part": "petal",
133 "phase": random.uniform(0, math.pi * 2)
134 })
135
136 # 2. 翠绿花托萼片与花茎 (约 20% 粒子)
137 stem_points = int(total_points * 0.20)
138 for _ in range(stem_points):
139 v = random.uniform(0, 1)
140 theta = random.uniform(0, 2 * math.pi)
141
142 if v <0.35: # 萼片 (5片包裹在花底的自然绿萼)
143 leaf_angle = theta * 5.0
144 r_sepal = (1.0 - v / 0.35) * 55.0 * (abs(math.sin(leaf_angle)) ** 0.6)
145 base_x = r_sepal * math.cos(theta)
146 base_z = r_sepal * math.sin(theta)
147 base_y = -22.0 - v * 35.0
148 col = random.choice(pal["sepal"])
149 size = 1
150 else: # 花茎 (微微弯曲的自然枝干)
151 t_stem = (v - 0.35) / 0.65
152 r_stem = random.uniform(2.5, 6.0)
153 bend_x = math.sin(t_stem * 3.5) * 10.0
154 bend_z = math.cos(t_stem * 2.8) * 6.0
155 base_x = r_stem * math.cos(theta) + bend_x
156 base_z = r_stem * math.sin(theta) + bend_z
157 base_y = -35.0 - t_stem * 95.0
158 col = random.choice(pal["stem"])
159 size = 1
160
161 particles.append({
162 "x": base_x, "y": base_y, "z": base_z,
163 "col": col, "size": size, "part": "stem",
164 "phase": random.uniform(0, math.pi * 2)
165 })
166
167 # 3. 周围弥漫的浪漫光斑 / 星尘 (约 8% 粒子)
168 ember_points = int(total_points * 0.08)
169 for _ in range(ember_points):
170 rx = random.uniform(-CUBE_SIZE, CUBE_SIZE)
171 ry = random.uniform(-CUBE_SIZE, CUBE_SIZE)
172 rz = random.uniform(-CUBE_SIZE, CUBE_SIZE)
173 col = random.choice(pal["edge"])
174 particles.append({
175 "x": rx, "y": ry, "z": rz,
176 "col": col, "size": 1, "part": "ember",
177 "phase": random.uniform(0, math.pi * 2)
178 })
179
180 return particles

198]

296 def project_3d(self, x, y, z, cx, cy):
297 """3D 到 2D 透视几何变换"""
298 cos_x, sin_x = math.cos(self.rot_x), math.sin(self.rot_x)
299 y1 = y * cos_x - z * sin_x
300 z1 = y * sin_x + z * cos_x
301
302 cos_y, sin_y = math.cos(self.rot_y), math.sin(self.rot_y)
303 x2 = x * cos_y + z1 * sin_y
304 z2 = -x * sin_y + z1 * cos_y
305
306 cam_dist = 450
307 depth = z2 + cam_dist
308 if depth <=15:
309 depth = 15
310 factor = (FOV * self.zoom) / depth
311 screen_x = cx + x2 * factor
312 screen_y = cy - y1 * factor
313 return screen_x, screen_y, z2

315 def draw_code_aesthetic(self):
316 """左侧浪漫极客源码背景 (视频同款排版)"""
317 lines = [

320 "const ripple = sin(15 * t) / 140.0;",
321 "const u = 1 - pow(1 - (3.4 * t % 2PI)/PI, 4)/2 + ripple;",
322 "const y_curl = 2 * pow(x*x - x, 2) * sin(p);",
323 "const r = u * (x * sin(p) + y_curl * cos(p));",
324 "const h = u * (x * cos(p) - y_curl * sin(p));",

344 def render_loop(self):
345 if not self.is_running:
346 return
347
348 t = time.time() - self.start_time
349 pal = PALETTES[current_theme]
350
351 if self.auto_rotate and not self.is_dragging:
352 self.rot_y += 0.011
353 self.rot_x = 0.45 + 0.08 * math.sin(t * 0.5)
354
355 # 优雅平缓的呼吸律动
356 pulse = 1.0 + 0.045 * math.sin(t * 2.4) * (abs(math.cos(t * 2.4)) ** 0.5)
357
358 self.canvas.delete("all")
359
360 # 1. 源码装饰
361 if self.show_code:
362 self.draw_code_aesthetic()
363
364 center_x = WIDTH * 0.60
365 center_y = HEIGHT * 0.50
366
367 # 2. 3D 立方体边框
368 proj_cube = []
369 for vx, vy, vz in CUBE_VERTICES:
370 sx, sy, sz = self.project_3d(vx, vy, vz, center_x, center_y)
371 proj_cube.append((sx, sy, sz))
372
373 for i, j in CUBE_EDGES:
374 x1, y1, z1 = proj_cube[i]
375 x2, y2, z2 = proj_cube[j]
376 avg_z = (z1 + z2) / 2
377 edge_col = pal["cube"] if avg_z <0 else "#1c2635"
378 self.canvas.create_line(x1, y1, x2, y2, fill=edge_col, width=1)
379
380 # 3. 收集并进行深度排序 (Painter's Algorithm 画家算法，保证前排花瓣完美遮挡后排花瓣)
381 render_queue = []
382 for p in self.particles:
383 if p["part"] == "petal":
384 wobble = 1.0 + 0.02 * math.sin(t * 3.0 + p["phase"])
385 px = p["x"] * pulse * wobble
386 py = p["y"] * pulse * wobble
387 pz = p["z"] * pulse * wobble
388 elif p["part"] == "stem":
389 px = p["x"] * (1.0 + 0.015 * pulse)
390 py = p["y"]
391 pz = p["z"] * (1.0 + 0.015 * pulse)
392 else:
393 px = p["x"] + math.sin(t * 1.2 + p["phase"]) * 5.0
394 py = p["y"] + math.cos(t * 1.2 + p["phase"]) * 5.0
395 pz = p["z"]
396
397 sx, sy, sz = self.project_3d(px, py, pz, center_x, center_y)
398 render_queue.append((sz, sx, sy, p["col"], p["size"]))
399
400 # 从深到浅排序渲染 (深度缓冲)
401 render_queue.sort(key=lambda item: item[0], reverse=True)
402
403 for sz, sx, sy, col, size in render_queue:
404 if sz <-80 and size >1:
405 size = 1
406 elif sz >70 and size == 1 and random.random() <0.12:
407 size = 2
408
409 if size == 1:
410 self.canvas.create_rectangle(sx, sy, sx+1, sy+1, outline=col, fill=col)
411 else:
412 self.canvas.create_rectangle(sx-1, sy-1, sx+1, sy+1, outline=col, fill=col)
413
414 # 4. HUD 交互提示
415 info_text = f"🌹 主题: {pal['name']} | [C] 切换配色 | [空格] 暂停旋转 | [+/-] 缩放 | 🖱️ 鼠标拖拽 360° 旋转"
416 self.canvas.create_text(
417 WIDTH // 2, HEIGHT - 22, text=info_text,
418 fill="#5a6b82", font=("Helvetica", 11)
419 )
420
421 try:
422 self.root.after(16, self.render_loop)

424 pass
